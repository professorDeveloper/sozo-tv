package com.saikou.sozo_tv.audit

import android.graphics.Rect
import android.os.Bundle
import android.widget.TextView
import androidx.appcompat.app.AppCompatActivity
import androidx.fragment.app.DialogFragment
import androidx.lifecycle.lifecycleScope
import com.saikou.sozo_tv.adapters.VideoServersAdapter
import com.saikou.sozo_tv.presentation.screens.play.VideoServerDialog
import com.saikou.sozo_tv.presentation.screens.play.dialog.SubtitleStyleDialog
import kotlinx.coroutines.delay
import kotlinx.coroutines.launch
import java.io.File

class UiAuditActivity : AppCompatActivity() {
    override fun onCreate(savedInstanceState: Bundle?) {
        super.onCreate(savedInstanceState)
        setContentView(TextView(this).apply { text="Sozo TV UI verification"; textSize=24f })
        val count=intent.getIntExtra("count",12)
        val dialog: DialogFragment = when(intent.getStringExtra("kind")) {
            "style" -> SubtitleStyleDialog()
            "popup" -> com.saikou.sozo_tv.presentation.screens.play.dialog.PlayerSettingsPopup().apply {
                rootRows = { List(count) { com.saikou.sozo_tv.adapters.SettingRow("Setting ${it+1}", "A long descriptive selected value for testing") } }
            }
            "quality" -> com.saikou.sozo_tv.presentation.screens.play.VideoQualityDialog(List(count) {
                com.saikou.sozo_tv.parser.models.VideoOption("", "A very long fansub provider name for UI verification", "1080p", com.saikou.sozo_tv.parser.models.AudioType.SUB, "High Quality extended metadata", false, fullText="")
            }, 0)
            "chooser" -> com.saikou.sozo_tv.presentation.screens.play.dialog.SubtitleChooserDialog.newInstance(emptyList(), null, false)
            "search" -> com.saikou.sozo_tv.presentation.screens.play.dialog.SubtitleSearchDialog.newInstance("",false,null,null)
            else -> VideoServerDialog(
            List(count) { VideoServersAdapter.ServerRow(if(it==0) "Animeav1 [SUB:HLS]" else "Server ${it+1} — extended title for TV layout verification", "1080p · 720p · 480p") },
            intent.getIntExtra("selected",0),
        )
        }
        dialog.show(supportFragmentManager,"audit")
        lifecycleScope.launch {
            delay(1500)
            val v=dialog.dialog?.window?.decorView?.findFocus()
            val r=Rect()
            v?.getGlobalVisibleRect(r)
            val width=v?.let { it.width*it.scaleX } ?: 0f
            val height=v?.let { it.height*it.scaleY } ?: 0f
            val ok=width>0 && r.width()+2>=width && r.height()+2>=height
            File(filesDir,"ui-audit.txt").writeText("${if(ok) "PASS" else "FAIL"}: focused expected=${width}x${height}; visible=${r.width()}x${r.height()}; rect=$r; class=${v?.javaClass?.simpleName}")
        }
    }
}
