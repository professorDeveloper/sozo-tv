package com.saikou.sozo_tv.audit

import android.os.Bundle
import android.widget.TextView
import androidx.appcompat.app.AppCompatActivity
import androidx.lifecycle.lifecycleScope
import com.saikou.sozo_tv.domain.model.SearchModel
import com.saikou.sozo_tv.domain.repository.SearchRepository
import com.saikou.sozo_tv.domain.repository.SearchSourceLeg
import com.saikou.sozo_tv.presentation.viewmodel.SearchViewModel
import kotlinx.coroutines.*
import kotlinx.coroutines.flow.*
import java.io.File

class SearchRaceActivity : AppCompatActivity() {
    override fun onCreate(savedInstanceState: Bundle?) {
        super.onCreate(savedInstanceState)
        val label = TextView(this).apply { textSize=22f; setPadding(48,48,48,48) }
        setContentView(label)
        lifecycleScope.launch {
            val repo = object : SearchRepository {
                override suspend fun searchAnime(query: String): Result<List<SearchModel>> {
                    // Simulate a provider that does not cooperate with cancellation.
                    withContext(NonCancellable) { delay(if(query=="Naruto A") 700 else 100) }
                    return if(query=="error") Result.failure(IllegalStateException("offline"))
                    else Result.success(listOf(SearchModel(1,query,null,null,null,null)))
                }
                override suspend fun searchMovie(query: String) = searchAnime(query)
                override fun searchAllSources(query: String): Flow<SearchSourceLeg> = emptyFlow()
            }
            val vm=SearchViewModel(repo)
            vm.searchAnime("Naruto A")
            delay(50)
            vm.searchAnime("Naruto B")
            delay(1000)
            val title=vm.searchResults.value?.firstOrNull()?.title
            val race="${if(title=="Naruto B") "PASS" else "FAIL"}: newest query wins; actual=$title"
            vm.searchAnime("error")
            delay(300)
            val error="${if(!vm.loading.value) "PASS" else "FAIL"}: failed request stops loading; actual=${vm.loading.value}"
            File(filesDir,"search-race.txt").writeText("$race\n$error\n")
            label.text="$race\n\n$error"
        }
    }
}
